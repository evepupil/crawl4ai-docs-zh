## URL 种子发现

智能 URL 发现技术，助力高效大规模爬取。即时发现数千个 URL，按相关性筛选，仅抓取重要内容。

### URL 种子发现 vs 深度爬取

```python

from crawl4ai import AsyncWebCrawler, CrawlerRunConfig
```

```python
import asyncio
from crawl4ai import AsyncUrlSeeder, SeedingConfig

async def basic_discovery():
    # 上下文管理器自动处理清理工作
    async with AsyncUrlSeeder() as seeder:
        
        # 从站点地图简单发现
        config = SeedingConfig(source="sitemap")

        print(f"从站点地图发现了 {len(urls)} 个 URL")
        for url in urls[:5]:
            print(f"  - {url['url']} (状态: {url['status']})")

# 手动清理（如果需要）
async def manual_cleanup():
    seeder = AsyncUrlSeeder()

        config = SeedingConfig(source="cc")  # Common Crawl
        urls = await seeder.urls("example.com", config)

        await seeder.close()

asyncio.run(basic_discovery())
```

### 元数据提取与分析

```python
async def metadata_extraction():
    async with AsyncUrlSeeder() as seeder:
        config = SeedingConfig(
            source="sitemap",
            extract_head=True,        # 提取 <head> 元数据
            pattern="*/blog/*",
            max_urls=50
        )
        
        urls = await seeder.urls("example.com", config)
        
        # 分析提取的元数据
        for url in urls[:5]:
            head_data = url['head_data']
            print(f"\nURL: {url['url']}")
            print(f"标题: {head_data.get('title', '无标题')}")
            
            # 标准元标签
            meta = head_data.get('meta', {})
            print(f"描述: {meta.get('description', '无')}")
            print(f"关键词: {meta.get('keywords', '无')}")
            print(f"作者: {meta.get('author', '无')}")
            
            # Open Graph 数据
            print(f"OG 图片: {meta.get('og:image', '无')}")
            print(f"OG 类型: {meta.get('og:type', '无')}")
            
            # JSON-LD 结构化数据
            jsonld = head_data.get('jsonld', [])
            if jsonld:
                print(f"结构化数据: {len(jsonld)} 项")
                for item in jsonld[:2]:
                    if isinstance(item, dict):
                        print(f"  类型: {item.get('@type', '未知')}")
                        print(f"  名称: {item.get('name', '无')}")

# 按元数据过滤
async def metadata_filtering():
    async with AsyncUrlSeeder() as seeder:
        config = SeedingConfig(
            source="sitemap",
            extract_head=True,
            max_urls=100
        )
        
        urls = await seeder.urls("news.example.com", config)
        
        # 按发布日期过滤（从 JSON-LD 获取）
        from datetime import datetime, timedelta
        recent_cutoff = datetime.now() - timedelta(days=7)
        
        recent_articles = []
        for url in urls:
            for jsonld in url['head_data'].get('jsonld', []):
                if isinstance(jsonld, dict) and 'datePublished' in jsonld:
                    try:
                        pub_date = datetime.fromisoformat(
                            jsonld['datePublished'].replace('Z', '+00:00')
                        )
                        if pub_date > recent_cutoff:
                            recent_articles.append(url)
                            break
                    except:
                        continue
        
        print(f"发现了 {len(recent_articles)} 篇近期文章")
```

```python
async def relevance_scoring():
    async with AsyncUrlSeeder() as seeder:
        # 查找关于 Python 异步编程的页面
        config = SeedingConfig(
            source="sitemap",
            extract_head=True,              # 基于内容的评分需要此选项
            query="python async await concurrency",
            scoring_method="bm25",
            score_threshold=0.3,            # 仅保留相关性 30%+ 的页面
            max_urls=20
        )
        
        urls = await seeder.urls("docs.python.org", config)
        
        # 结果自动按相关性排序
        print("最相关的 Python 异步内容:")
        for url in urls[:5]:
            score = url['relevance_score']
            title = url['head_data'].get('title', '无标题')
            print(f"[{score:.2f}] {title}")
            print(f"        {url['url']}")

async def complex_queries():

            )
            
            urls = await seeder.urls("learning-site.com", config)
```
