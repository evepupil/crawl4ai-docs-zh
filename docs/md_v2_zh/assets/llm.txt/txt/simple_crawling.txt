## 简易爬取

使用 AsyncWebCrawler 进行基础网络爬取操作，包括配置和响应处理。

### 基础设置

```python
import asyncio
from crawl4ai import AsyncWebCrawler, BrowserConfig, CrawlerRunConfig

async def main():
    browser_config = BrowserConfig()  # 默认浏览器设置
    run_config = CrawlerRunConfig()   # 默认爬取设置

    async with AsyncWebCrawler(config=browser_config) as crawler:
        result = await crawler.arun(
            url="https://example.com",
            config=run_config
        )
        print(result.markdown)

if __name__ == "__main__":
    asyncio.run(main())
```

### 理解 CrawlResult

```python
from crawl4ai.markdown_generation_strategy import DefaultMarkdownGenerator
from crawl4ai.content_filter_strategy import PruningContentFilter

config = CrawlerRunConfig(
    markdown_generator=DefaultMarkdownGenerator(
        content_filter=PruningContentFilter(threshold=0.6),
        options={"ignore_links": True}
    )
)

result = await crawler.arun("https://example.com", config=config)

# 不同内容格式
print(result.html)                    # 原始 HTML
print(result.cleaned_html)            # 清理后的 HTML  
print(result.markdown.raw_markdown)   # 原始 Markdown
print(result.markdown.fit_markdown)   # 过滤后的 Markdown

# 状态信息
print(result.success)      # True/False
print(result.status_code)  # HTTP 状态码 (200, 404 等)

# 提取的内容
print(result.media)        # 图片、视频、音频
print(result.links)        # 内部/外部链接
```

### 完整示例

```python
import asyncio
from crawl4ai import AsyncWebCrawler, BrowserConfig, CrawlerRunConfig, CacheMode

async def comprehensive_crawl():
    browser_config = BrowserConfig(verbose=True)
    
    run_config = CrawlerRunConfig(
        # 内容过滤
        word_count_threshold=10,
        excluded_tags=['form', 'header', 'nav'],
        exclude_external_links=True,
        
        # 内容处理
        process_iframes=True,
        remove_overlay_elements=True,
        
        # 缓存控制
        cache_mode=CacheMode.ENABLED
    )

    async with AsyncWebCrawler(config=browser_config) as crawler:
        result = await crawler.arun(
            url="https://example.com",
            config=run_config
        )
        
        if result.success:
            # 显示内容摘要
            print(f"标题: {result.metadata.get('title', '无标题')}")
            print(f"内容: {result.markdown[:500]}...")
            
            # 处理媒体
            images = result.media.get("images", [])
            print(f"找到 {len(images)} 张图片")
            for img in images[:3]:  # 前 3 张图片
                print(f"  - {img.get('src', '无来源')}")
            
            # 处理链接
            internal_links = result.links.get("internal", [])
            print(f"找到 {len(internal_links)} 个内部链接")
            for link in internal_links[:3]:  # 前 3 个链接
                print(f"  - {link.get('href', '无链接')}")
                
        else:
            print(f"❌ 爬取失败: {result.error_message}")
            print(f"状态: {result.status_code}")

if __name__ == "__main__":
    asyncio.run(comprehensive_crawl())
```

## 表格提取

从 HTML 表格中自动检测和评分，提取结构化数据。

### 基础表格提取

```python
import asyncio
import pandas as pd
from crawl4ai import AsyncWebCrawler, CrawlerRunConfig, CacheMode

async def extract_tables():
    async with AsyncWebCrawler() as crawler:
        config = CrawlerRunConfig(
            table_score_threshold=7,  # 数值越高 = 检测越严格
            cache_mode=CacheMode.BYPASS
        )
        
        result = await crawler.arun("https://example.com/tables", config=config)
        
        if result.success and result.tables:
            # 新表格字段 (v0.6+)
            for i, table in enumerate(result.tables):
                print(f"表格 {i+1}:")
                print(f"表头: {table['headers']}")
                print(f"行数: {len(table['rows'])}")
                print(f"标题: {table.get('caption', '无标题')}")
                
                # 转换为 DataFrame
                df = pd.DataFrame(table['rows'], columns=table['headers'])
                print(df.head())

asyncio.run(extract_tables())
```

### 高级表格处理

```python
from crawl4ai import LXMLWebScrapingStrategy

async def process_financial_tables():
    config = CrawlerRunConfig(
        table_score_threshold=8,  # 对数据表格进行严格检测
        scraping_strategy=LXMLWebScrapingStrategy(),
        keep_data_attributes=True,
        scan_full_page=True
    )
    
    async with AsyncWebCrawler() as crawler:
        result = await crawler.arun("https://coinmarketcap.com", config=config)
        
        if result.tables:
            # 获取主数据表（通常是第一个/最大的）
            main_table = result.tables[0]
            
            # 创建 DataFrame
            df = pd.DataFrame(
                main_table['rows'],
                columns=main_table['headers']
            )
            
            # 清理和处理数据
            df = clean_financial_data(df)
            
            # 保存用于分析
            df.to_csv("market_data.csv", index=False)
            return df

def clean_financial_data(df):
    """清理货币符号、百分比和大数字"""
    for col in df.columns:
        if 'price' in col.lower():
            # 移除货币符号
            df[col] = df[col].str.replace(r'[^\d.]', '', regex=True)
            df[col] = pd.to_numeric(df[col], errors='coerce')
        
        elif '%' in str(df[col].iloc[0]):
            # 转换百分比
            df[col] = df[col].str.replace('%', '').astype(float) / 100
        
        elif any(suffix in str(df[col].iloc[0]) for suffix in ['B', 'M', 'K']):
            # 处理大数字（十亿、百万等）
            df[col] = df[col].apply(convert_large_numbers)
    
    return df

def convert_large_numbers(value):
    """转换 1.5B -> 1500000000"""
    if pd.isna(value):
        return float('nan')
    
    value = str(value)
    multiplier = 1
    if 'B' in value:
        multiplier = 1e9
    elif 'M' in value:
        multiplier = 1e6
    elif 'K' in value:
        multiplier = 1e3
    
    number = float(re.sub(r'[^\d.]', '', value))
    return number * multiplier
```

### 多表格处理

```python
async def extract_all_tables():
    async with AsyncWebCrawler() as crawler:
        result = await crawler.arun("https://example.com/data", config=config)
        
        tables_data = {}
        
        for i, table in enumerate(result.tables):
            # 基于内容创建有意义的名称
            table_name = (
                table.get('caption') or 
                f"table_{i+1}_{table['headers'][0]}"
            ).replace(' ', '_').lower()
            
            df = pd.DataFrame(table['rows'], columns=table['headers'])
            
            # 存储元数据
            tables_data[table_name] = {
                'dataframe': df,
                'headers': table['headers'],
                'row_count': len(table['rows']),
                'caption': table.get('caption'),
                'summary': table.get('summary')
            }
        
        return tables_data

# 用法
tables = await extract_all_tables()
for name, data in tables.items():
    print(f"{name}: {data['row_count']} 行")
    data['dataframe'].to_csv(f"{name}.csv")
```

# 10: 仅完美数据表格（表头 + 数据行）
