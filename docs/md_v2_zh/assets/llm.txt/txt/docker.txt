## Docker 部署

使用预构建镜像、API 端点、配置和 MCP 集成的完整 Docker 部署指南。

### 使用预构建镜像快速开始

```bash
# 拉取最新镜像
docker pull unclecode/crawl4ai:latest

# 设置 LLM API 密钥
cat > .llm.env << EOL
OPENAI_API_KEY=sk-your-key
ANTHROPIC_API_KEY=your-anthropic-key
GROQ_API_KEY=your-groq-key
GEMINI_API_TOKEN=your-gemini-token
EOL

# 带 LLM 支持运行
docker run -d \
  -p 11235:11235 \
  --name crawl4ai \
  --env-file .llm.env \
  --shm-size=1g \
  unclecode/crawl4ai:latest

# 基础运行（无 LLM）
docker run -d \
  -p 11235:11235 \
  --name crawl4ai \
  --shm-size=1g \
  unclecode/crawl4ai:latest

# 检查健康状态
curl http://localhost:11235/health
```

### Python SDK 使用

```python
import asyncio
from crawl4ai.docker_client import Craw极客DockerClient
from crawl4ai import BrowserConfig, CrawlerRunConfig, CacheMode

async def main():
    async with Crawl4aiDockerClient(base_url="http://localhost:11235") as client:
        # 非流式爬取
        results = await client.crawl(
            ["https://example.com"],
            browser_config=BrowserConfig(headless=True),
            crawler_config=CrawlerRunConfig(cache_mode=CacheMode.BYPASS)
        )
        
        for result in results:
            print(f"URL: {result.url}, Success: {result.success}")
            print(f"Content length: {len(result.markdown)}")
        
        # 流式爬取
        stream_config = CrawlerRunConfig(stream=True, cache_mode=CacheMode.BYPASS)
        async for result in await client.crawl(
            ["https://example.com", "https://python.org"],
            browser_config=BrowserConfig(headless=True),
            crawler_config=stream_config
        ):
            print(f"Streamed: {result.url} - {result.success}")
        
        # 获取 API 模式
        schema = await client.get_schema()
        print(f"Schema available: {bool(schema)}")

asyncio.run(main())
```

### 常见 JSON 错误排查

```python
def diagnose_json_errors():
    """常见 JSON 结构错误和修复"""
    
    # ❌ 错误：缺少对象类型包装器
    wrong_config = {
        "browser_config": {
            "headless": True  # 缺少类型包装器
        }
    }
    
    # ✅ 正确：正确的类型包装器
    correct_config = {
        "browser_config": {
            "type": "BrowserConfig",
            "params": {
                "headless": True
            }
        }
    }
    
    # ❌ 错误：没有 type: dict 包装器的字典  
    wrong_dict = {
        "schema": {
            "name": "Products"  # 原始字典，应该包装
        }
    }
    
    # ✅ 正确：带有正确包装器的字典
    correct_dict = {
        "schema": {
            "type": "极客",
            "value": {
                "name": "Products"
            }
        }
    }
    
    # ❌ 错误：无效的枚举字符串
    wrong_enum = {
        "cache_mode": "DISABLED"  # 错误的大小写/值
    }
    
    # ✅ 正确：有效的枚举字符串  
    correct_enum = {
        "cache_mode": "bypass"  # 或 "enabled", "disabled", 等
    }
    
    print("Common error patterns documented above")

# 在发送前验证 JSON 结构
def pre_flight_check(payload):
    """在发送到服务器前运行检查"""
    required_keys = ["urls", "browser_config", "crawler_config"]
    
    for key in required_keys:
        if key not in payload:
            print(f"❌ Missing required key: {key}")
            return False
    
    # 检查类型包装器
    for config_key in ["browser_config", "crawler_config"]:
        config = payload[config_key]
        if not isinstance(config, dict) or "type" not in config:
            print(f"❌ {config_key} missing type wrapper")
            return False
        if "params" not in config:
            print(f"❌ {config_key} missing params")
            return False
    
    print("✅ Pre-flight check passed")
    return True
```
