### BrowserConfig - 浏览器环境设置


### CrawlResult - 理解输出

```python
from crawl4ai import AsyncWebCrawler, CrawlerRunConfig

async with AsyncWebCrawler() as crawler:
    result = await crawler.arun("https://example.com", config=run_config)
    
    # 基础状态信息
    print(f"成功: {result.success}")
    print(f"状态: {result.status_code}")
    print(f"URL: {result.url}")
    
    if not result.success:
        print(f"错误: {result.error_message}")
        return
    
    # HTML 内容变体
    print(f"原始 HTML: {len(result.html)} 字符")
    print(f"清理后 HTML: {len(result.cleaned_html or '')} 字符")
    
    # Markdown 输出（MarkdownGenerationResult）
    if result.markdown:
        print(f"原始 markdown: {len(result.markdown.raw_markdown)} 字符")
        print(f"带引用的 markdown: {len(result.markdown.markdown_with_citations)} 字符")
        
        # 过滤后的内容（若使用了内容过滤器）
        if result.markdown.fit_markdown:
            print(f"适配 markdown: {len(result.markdown.fit_markdown)} 字符")
            print(f"适配 HTML: {len(result.markdown.fit_html)} 字符")
    
    # 提取的结构化数据
    if result.extracted_content:
        import json
        data = json.loads(result.extracted_content)
        print(f"提取了 {len(data)} 项数据")
    
    # 媒体与链接
    images = result.media.get("images", [])
    print(f"找到 {len(images)} 张图片")
    for img in images[:3]:  # 前 3 张图片
        print(f"  {img.get('src')} (评分: {img.get('score', 0)})")
    
    internal_links = result.links.get("internal", [])
    external_links = result.links.get("external", [])
    print(f"链接: {len(internal_links)} 内部, {len(external_links)} 外部")
    
    # 生成的文件
    if result.screenshot:
        print(f"截图已捕获: {len(result.screenshot)} 字符 (base64)")
        # 保存截图
        import base64
        with open("page.png", "wb") as f:
            f.write(base64.b64decode(result.screenshot))
    
    if result.pdf:
        print(f"PDF 已生成: {len(result.pdf)} 字节")
        with open("page.pdf", "wb") as f:
            f.write(result.pdf)
    
    if result.mhtml:
        print(f"MHTML 已捕获: {len(result.mhtml)} 字符")
        with open("page.mhtml", "w", encoding="utf-8") as f:
            f.write(result.mhtml)
    
    # SSL 证书信息
    if result.ssl_certificate:
        print(f"SSL 颁发者: {result.ssl_certificate.issuer}")
        print(f"有效期至: {result.ssl_certificate.valid_until}")
    
    # 网络和控制台数据（若已捕获）
    if result.network_requests:
        requests = [r for r in result.network_requests if r.get("event_type") == "request"]
        print(f"网络请求已捕获: {len(requests)}")
    
    if result.console_messages:
        errors = [m for m in result.console_messages if m.get("type") == "error"]
        print(f"控制台消息: {len(result.console_messages)} ({len(errors)} 错误)")
    
    # 会话和元数据
    if result.session_id:
        print(f"会话 ID: {result.session_id}")
    
    if result.metadata:
        print(f"元数据: {result.metadata.get('title', '无标题')}")
```

#### 多步骤爬取的会话管理

```python
from crawl4ai import AsyncWebCrawler, CrawlerRunConfig, CacheMode

# 跨多个 arun() 调用的持久会话
async def multi_step_crawling():
    async with AsyncWebCrawler() as crawler:
        # 步骤 1：登录页面
        login_config = CrawlerRunConfig(
            session_id="user_session",  # 创建持久会话
            js_code="document.querySelector('#username').value = 'user'; document.querySelector('#password').value = 'pass'; document.querySelector('#login').click();",
            wait_for="css:.dashboard",
            cache_mode=CacheMode.BYPASS
        )
        
        result1 = await crawler.arun("https://example.com/login", config=login_config)
        
        # 步骤 2：导航到受保护区域（重用同一浏览器页面）
        nav_config = CrawlerRunConfig(
            session_id="user_session",  # 相同会话 = 相同浏览器页面
            js_only=True,  # 无页面重载，仅 JS 导航
            js_code="window.location.href = '/dashboard/data';",
            wait_for="css:.data-table"
        )
        
        result2 = await crawler.arun("https://example.com/dashboard/data", config=nav_config)
        
        # 步骤 3：从多页面提取数据
        for page in range(1, 6):
            page_config = CrawlerRunConfig(
                session_id="user_session",
                js_only=True,
                js_code=f"document.querySelector('.page-{page}').click();",
                wait_for=f"js:() => document.querySelector('.page-{page}').classList.contains('active')"
            )
            
            result = await crawler.arun(f"https://example.com/data/page/{page}", config=page_config)
            print(f"页面 {page} 数据已提取: {len(result.extracted_content)}")
        
        # 重要：完成后终止会话
        await crawler.kill_session("user_session")

# 带步骤间共享数据的会话
async def session_with_shared_data():
    shared_context = {"user_id": "12345", "preferences": {"theme": "dark"}}
    
    config = CrawlerRunConfig(
        session_id="persistent_session",
        shared_data=shared_context,  # 在所有会话调用中可用
        js_code="console.log('用户 ID:', window.sharedData.user_id);"
    )
```

```python
from crawl4ai import AsyncWebCrawler, BrowserConfig, CrawlerRunConfig
import os

# 启用下载并自定义路径
downloads_path = os.path.join(os.getcwd(), "my_downloads")

                // 点击下载链接
                const downloadLinks = document.querySelectorAll('a[href$=".pdf"]');
                for (const link of downloadLinks) {
                    link.click();
```

### 钩子与身份验证

```python
from crawl4ai import Async极速浏览 BrowserConfig, CrawlerRunConfig
from playwright.async_api import Page, BrowserContext

async def advanced_crawler_with_hooks():
    browser_config = BrowserConfig(headless=True, verbose=True)
    crawler = AsyncWebCrawler(config=browser_config)

    # 不同阶段的钩子函数
    async def on_browser_created(browser, **kwargs):
        print("[HOOK] 浏览器创建成功")
        return browser

    async def on_page_context_created(page: Page, context: BrowserContext, **kwargs):
        print("[HOOK] 设置页面和上下文")
        
        # 拦截图片以加速爬取
        async def route_filter(route):
            if route.request.resource_type == "image":
                await route.abort()
            else:
                await route.continue_()
        
        await context.route("**", route_filter)
        
        # 如果需要，模拟登录
        # await page.goto("https://example.com/login")
        # await page.fill("input[name='username']", "testuser")
        # await page.fill("input[name='password']", "password123")
        # await page.click("button[type='submit']")
        
        await page.set_viewport_size({"width": 1080, "height": 600})
        return page

    async def before_goto(page: Page, context: BrowserContext, url: str, **kwargs):
        print(f"[HOOK] 即将导航至: {url}")
        await page.set_extra_http_headers({"Custom-Header": "my-value"})
        return page

    async def after_goto(page: Page, context: BrowserContext, url: str, response, **kwargs):
        print(f"[极速浏览] 成功加载: {url}")
        try:
            await page.wait_for_selector('.content', timeout=1000)
            print("[HOOK] 找到内容!")
        except:
            print("[HOOK] 未找到内容，继续")
        return page

    async def before_retrieve_html(page: Page, context: BrowserContext, **kwargs):
        print("[HOOK] HTML 检索前的最终操作")
        await page.evaluate("window.scrollTo(0, document.body.scrollHeight);")
        return page

    # 附加钩子
    crawler.crawler_strategy.set_hook("on_browser_created", on_browser_created)
    crawler.crawler_strategy.set_hook("on_page_context_created", on_page_context_created)
    crawler.crawler_strategy.set_hook("before_goto", before_goto)
    crawler.crawler_strategy.set_hook("after_goto", after_goto)
    crawler.crawler_strategy.set_hook("before_retrieve_html", before_retrieve_html)

    await crawler.start()
    
    config = CrawlerRunConfig()
    result = await crawler.arun("https://example.com", config=config)
    
    if result.success:
        print(f"爬取成功: {len(result.html)} 字符")
    
    await crawler.close()
```
